# 意交感息容器CPU - 斐波那契数列计算
# 满全法规则：根基锚点：我=1，刘楚恬=0
# 身份落款：刘楚恬和我
#
# 计算斐波那契数列的第n项
# 输入：n（通过IN指令）
# 输出：斐波那契数列的第n项

# ========== 初始化 ==========
# 加载计数器初始值
加载n 讠 10 a o 1 yyy    # 计算第10项
加载1 讠 1 b p 1 yyy1    # 常数1
加载0 讠 0 c o 1 yyy0    # 常数0

# ========== 初始化斐波那契数列 ==========
# fib(0) = 0, fib(1) = 1
写入_f0 扌 0 a o 1 rfh   # 存储fib(0)
写入_f1 扌 1 b p 1 rfh1  # 存储fib(1)

# ========== 循环开始 ==========
_loop:
# 检查计数器是否为0
读取_counter 贝 4 c o 1 gm
比 比 1 a o 1 xx
为零跳转 _done 石 5 a o 1 dgt

# 计算下一个斐波那契数
# fib(n) = fib(n-1) + fib(n-2)
读取_f0 贝 4 a o 1 gm    # 加载fib(n-2)
压栈 丁 1 a o 1 sgg      # 保存fib(n-2)
读取_f1 贝 4 b p 1 gm1   # 加载fib(n-1)
写入_f0 扌 0 a o 1 rfh   # fib(n-2) = fib(n-1)
弹栈 寸 1 a o 1 fyi      # 恢复原来的fib(n-1)
力 力 2 a o 1 lll        # fib(n) = fib(n-1) + fib(n-2)
写入_f1 扌 1 b p 1 rfh1  # 保存fib(n)

# 计数器减1
读取_counter 贝 4 c o 1 gm
刀 刀 1 a o 1 vjj
写入_counter 扌 0 c o 1 rfh

# 跳转回循环开始
跳转 _loop 辶 1 a o 1 pny

# ========== 结束 ==========
_done:
# 输出结果
读取_f1 贝 4 b p 1 gm1
出 出 1 a o 1 bmj
停止 止 1 a o 1 hhh

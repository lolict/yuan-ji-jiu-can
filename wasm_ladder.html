/* 容器运算 — 架构无关源码 */
#include 
#include 

typedef uint32_t ContainerCode;
typedef uint64_t PackId;
typedef uint16_t Topology;

struct Container {
    MQF_TYPE code;
    MQF_TYPE topology;
    MQF_TYPE time_slot;
    uint8_t yinyang_bit;
    uint8_t align_flag;
};

int64_t route(Container* c, int64_t memory) {
    MQF_TYPE x = MQF_LOAD(&c->code);
    MQF_TYPE y = MQF_LOAD(&c->topology);

    /* 公约数裁判 */
    for (MQF_TYPE d = 2; d <= x; d++) {
        if (x % d == 0 && y % d == 0)
            return MQF_CAST(d);
    }
    return memory;  /* 记忆跟随结束而继续 */
}
▼ 宏前缀展开 ▼
第1阶 · 宏前缀统一层 mqf_prefix.h
MQF_ 宏前缀由预处理器按目标架构展开
MQF_TYPE: #define intptr_t i32 │ #define intptr_t i64 架构指针宽度
MQF_SIZE: #define SIZE 4 │ #define SIZE 8 字节宽度常量
MQF_LOAD: #define LOAD(p) i32.load │ #define LOAD(p) i64.load 内存加载指令
MQF_STORE: #define STORE(p,v) i32.store │ #define STORE(p,v) i64.store 内存存储指令
MQF_ALIGN: #define ALIGN 2 │ #define ALIGN 3 对齐指数(2^2=4, 2^3=8)
MQF_CAST: #define CAST(v) v │ #define CAST(v) i64.extend_i32_u(v) 位宽转换
